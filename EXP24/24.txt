LDA 2801H
MOV C,A

MOV A,B
CMP C
JNC START

MOV A,C
MOV B,A

START: MOV A,B
MOV D,A

LOOP: MOV A,D
ADD B
MOV D,A

MOV A,D
MOV E,A

CHECK: MOV A,E
CMP C
JZ DONE

JC NEXT

SUB C

ADD C

JMP CHECK
